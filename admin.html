<script>

const $=id=>document.getElementById(id);

const esc=s=>String(s??'').replace(/[&<>"']/g,c=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c]));

const eur=n=>Number(n).toLocaleString('de-DE',{style:'currency',currency:'EUR'});

let pw='',items=[],cur=null,isNew=false,dirty=false,newPics={};


async function api(body){

  const r=await fetch('/.netlify/functions/admin-save',{method:'POST',body:JSON.stringify({password:pw,...body})});

  const d=await r.json().catch(()=>({error:'Serverfehler'}));

  if(!r.ok) throw new Error(d.error||'Fehler');

  return d;

}

async function login(){

  pw=$('pw').value||pw; $('lerr').textContent='';

  try{items=(await api({action:'load'})).products;sessionStorage.setItem('pw',pw);$('login').hidden=true;$('app').hidden=false;render()}

  catch(e){$('lerr').textContent=e.message;sessionStorage.removeItem('pw')}

}

$('go').onclick=login;$('pw').onkeydown=e=>{if(e.key==='Enter')login()};

if(sessionStorage.getItem('pw')){pw=sessionStorage.getItem('pw');login()}


function setDirty(v){dirty=v;$('pub').disabled=!v;$('state').textContent=v?'Ungespeicherte Änderungen':''}

window.onbeforeunload=()=>dirty?'Es gibt nicht veröffentlichte Änderungen.':undefined;


function render(){

  const q=$('q').value.toLowerCase();

  $('list').innerHTML=items.map((i,k)=>({i,k})).filter(({i})=>(i.brand+' '+i.name+' '+i.cat).toLowerCase().includes(q)).map(({i,k})=>

    `<div class="row ${i.sold?'sold':''}"><img src="${newPics[i.folder]?.[0]||`images/${esc(i.folder)}/A1.jpeg`}" alt="">

    <div class="t"><b>${esc(i.brand)} – ${esc(i.name)}</b><small>${esc(i.cat)} · Gr. ${esc(i.size)} · ${eur(i.price)}${i.sold?' · VERKAUFT':''}</small></div>

    <button data-e="${k}">Bearbeiten</button><button data-d="${k}">Löschen</button></div>`).join('')||'<p>Keine Artikel gefunden.</p>';

}

$('q').oninput=render;

$('list').onclick=e=>{

  if(e.target.dataset.e!==undefined) edit(items[e.target.dataset.e],false);

  if(e.target.dataset.d!==undefined&&confirm('Artikel wirklich löschen?')){items.splice(e.target.dataset.d,1);setDirty(true);render()}

};

$('new').onclick=()=>{const id=Math.max(0,...items.map(i=>i.id))+1;edit({id,folder:'Art'+id,cat:'Pullover',brand:'',name:'',size:'',desc:'',price:15,sold:false},true)};


function edit(it,n){

  cur=it;isNew=n;

  ['brand','name','cat','size','desc','price'].forEach(k=>$(k).value=it[k]);

  $('sold').value=it.sold?'1':'0';

  newPics[it.folder]=newPics[it.folder]||[];

  $('pics').innerHTML=[0,1,2].map(i=>`<div class="pic"><img id="p${i}" src="${newPics[it.folder][i]||`images/${esc(it.folder)}/A${i+1}.jpeg`}" alt="" onerror="this.removeAttribute('src')">Foto ${i+1}${i?'':' (Titelbild)'}<input type="file" accept="image/*" data-n="${i}"></div>`).join('');

  $('dlg').showModal();

}

$('pics').onchange=e=>{

  const n=+e.target.dataset.n,file=e.target.files[0];if(!file)return;

  const im=new Image();

  im.onload=()=>{

    const s=Math.min(1,1200/Math.max(im.width,im.height)),c=document.createElement('canvas');

    c.width=im.width*s;c.height=im.height*s;c.getContext('2d').drawImage(im,0,0,c.width,c.height);

    const d=c.toDataURL('image/jpeg',.82);newPics[cur.folder][n]=d;$('p'+n).src=d;

  };

  im.src=URL.createObjectURL(file);

};

$('f').onsubmit=e=>{

  e.preventDefault();

  Object.assign(cur,{brand:$('brand').value.trim(),name:$('name').value.trim(),cat:$('cat').value,size:$('size').value.trim(),desc:$('desc').value.trim(),price:parseFloat($('price').value),sold:$('sold').value==='1'});

  if(isNew) items.push(cur);

  $('dlg').close();setDirty(true);render();

};

$('cancel').onclick=()=>$('dlg').close();


$('pub').onclick=async()=>{

  const images=[];

  Object.entries(newPics).forEach(([f,a])=>a.forEach((d,i)=>d&&images.push({folder:f,n:i+1,data:d.split(',')[1]})));

  $('pub').disabled=true;$('state').textContent='Wird gespeichert …';

  try{

    await api({action:'save',products:items,images});

    newPics={};dirty=false;$('state').textContent='✓ Gespeichert. In ca. 1–2 Minuten live.';

  }catch(e){$('state').textContent='Fehler: '+e.message;$('pub').disabled=false}

};

</script></body></html>
